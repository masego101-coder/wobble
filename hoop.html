function makePuzzle(){ const count=colourCount(); const cap=capacity(); // Preserve the beginner levels. if(progress.level===1){ return [[0,0,1],[1,1,0],[]]; } if(progress.level<=3){ const puzzle=Array.from( {length:count+1}, (_,i)=>ic===colour))continue; for(let to=0;to=cap)continue; if(from===lastTo&&to===lastFrom)continue; options.push([from,to]); } } if(!options.length)break; const [from,to]=options[ Math.floor(Math.random()*options.length) ]; puzzle[to].push(puzzle[from].pop()); lastFrom=from; lastTo=to; } return puzzle; } // Increasing scramble depth from Level 4. const level=progress.level; const steps=Math.min(500,95+(level-4)*24); function puzzleScore(puzzle){ let score=0; let mixedStacks=0; let transitions=0; for(const stack of puzzle){ if(!stack.length)continue; const colours=new Set(stack); const diversity=colours.size; if(diversity>1){ mixedStacks++; score+=45+(diversity*30)+(stack.length*12); } for(let i=1;i=3&&diversity>=3){ score+=35; } if(stack.length===2&&diversity===2){ score+=15; } } score+=mixedStacks*25; score+=transitions*12; return score; } let bestPuzzle=null; let bestScore=-Infinity; // Generate many candidates and select the most complex. for(let attempt=0;attempt<100;attempt++){ const puzzle=Array.from( {length:count+1}, (_,i)=>ic===colour))continue; for(let to=0;to=cap)continue; if(from===lastTo&&to===lastFrom)continue; options.push([from,to]); } } if(!options.length)break; const [from,to]=options[ Math.floor(Math.random()*options.length) ]; puzzle[to].push(puzzle[from].pop()); lastFrom=from; lastTo=to; } const score=puzzleScore(puzzle); if(score>bestScore){ bestScore=score; bestPuzzle=puzzle.map(stack=>stack.slice()); } } return bestPuzzle; }
