function makePuzzle(){ const count = colourCount(); const cap = capacity(); // Level 1: preserve the original beginner puzzle. if(progress.level === 1){ return [ [0,0,1], [1,1,0], [] ]; } // Levels 2 and 3: preserve introductory difficulty. if(progress.level <= 3){ const puzzle = Array.from( {length: count + 1}, (_,i) => i === count ? [] : Array(cap).fill(i) ); let lastFrom = -1; let lastTo = -1; const steps = 25 + progress.level * 9; for(let step = 0; step < steps; step++){ const options = []; for(let from = 0; from < puzzle.length; from++){ if(puzzle[from].length === 0) continue; const colour = puzzle[from][puzzle[from].length - 1]; if(!puzzle[from].every(c => c === colour)) continue; for(let to = 0; to < puzzle.length; to++){ if(to === from || puzzle[to].length >= cap) continue; if(from === lastTo && to === lastFrom) continue; options.push([from,to]); } } if(options.length === 0) break; const [from,to] = options[ Math.floor(Math.random() * options.length) ]; puzzle[to].push(puzzle[from].pop()); lastFrom = from; lastTo = to; } return puzzle; } // Level 4 onward: improved strategic puzzle generation. const level = progress.level; const steps = Math.min(500, 95 + (level - 4) * 24); function puzzleScore(puzzle){ let score = 0; let mixedStacks = 0; let transitions = 0; let buriedTransitions = 0; let occupiedSticks = 0; for(const stack of puzzle){ if(stack.length === 0) continue; occupiedSticks++; const diversity = new Set(stack).size; if(diversity > 1){ mixedStacks++; score += 45 + diversity * 30 + stack.length * 12; } for(let i = 1; i < stack.length; i++){ if(stack[i] !== stack[i-1]){ transitions++; score += 15; // Reward colour changes buried beneath other hoops. if(i < stack.length - 1){ buriedTransitions++; } } } if(stack.length === cap && diversity === 1){ score -= 35; } if(stack.length >= 3 && diversity >= 3){ score += 35; } if(stack.length === 2 && diversity === 2){ score += 15; } } score += mixedStacks * 25; score += transitions * 12; score += buriedTransitions * 18; score += occupiedSticks * 4; return score; } let bestPuzzle = null; let bestScore = -Infinity; // Compare 100 candidate puzzles. for(let attempt = 0; attempt < 100; attempt++){ const puzzle = Array.from( {length: count + 1}, (_,i) => i === count ? [] : Array(cap).fill(i) ); let lastFrom = -1; let lastTo = -1; for(let step = 0; step < steps; step++){ const options = []; for(let from = 0; from < puzzle.length; from++){ if(puzzle[from].length === 0) continue; const source = puzzle[from]; const colour = source[source.length - 1]; // The source must remain empty or have the same // colour exposed after the hoop is removed. if( source.length >= 2 && source[source.length - 2] !== colour ){ continue; } for(let to = 0; to < puzzle.length; to++){ if(to === from || puzzle[to].length >= cap) continue; // Avoid immediately undoing the previous scramble. if(from === lastTo && to === lastFrom) continue; options.push([from,to]); } } if(options.length === 0) break; const [from,to] = options[ Math.floor(Math.random() * options.length) ]; puzzle[to].push(puzzle[from].pop()); lastFrom = from; lastTo = to; } const score = puzzleScore(puzzle); if(score > bestScore){ bestScore = score; bestPuzzle = puzzle.map(stack => stack.slice()); } } return bestPuzzle; }
