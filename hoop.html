function makePuzzle(){ const count = colourCount(); const cap = capacity(); // Level 1: preserve the original beginner puzzle. if(progress.level === 1){ return [ [0,0,1], [1,1,0], [] ]; } // Levels 2 and 3: gentle introductory difficulty. if(progress.level <= 3){ const puzzle = Array.from( {length: count + 1}, (_,i) => i === count ? [] : Array(cap).fill(i) ); let lastFrom = -1; let lastTo = -1; const steps = 25 + progress.level * 9; for(let step = 0; step < steps; step++){ const options = []; for(let from = 0; from < puzzle.length; from++){ if(puzzle[from].length <= 0) continue; const colour = puzzle[from][puzzle[from].length - 1]; if(!puzzle[from].every(c => c === colour)) continue; for(let to = 0; to < puzzle.length; to++){ if(to === from || puzzle[to].length >= cap) continue; if(from === lastTo && to === lastFrom) continue; options.push([from,to]); } } if(options.length === 0) break; const [from,to] = options[ Math.floor(Math.random() * options.length) ]; puzzle[to].push(puzzle[from].pop()); lastFrom = from; lastTo = to; } return puzzle; } // Level 4 onward: progressively deeper puzzles. const level = progress.level; const steps = Math.min(500, 95 + (level - 4) * 24); function puzzleScore(puzzle){ let score = 0; let mixedStacks = 0; let transitions = 0; for(const stack of puzzle){ if(stack.length === 0) continue; const colours = new Set(stack); const diversity = colours.size; if(diversity > 1){ mixedStacks++; score += 45 + diversity * 30 + stack.length * 12; } for(let i = 1; i <= stack.length - 1; i++){ if(stack[i] !== stack[i-1]){ transitions++; score += 15; } } if(stack.length === cap && diversity === 1){ score -= 35; } if(stack.length >= 3 && diversity >= 3){ score += 35; } if(stack.length === 2 && diversity === 2){ score += 15; } } score += mixedStacks * 25; score += transitions * 12; return score; } let bestPuzzle = null; let bestScore = -Infinity; // Generate 100 candidate puzzles. for(let attempt = 0; attempt < 100; attempt++){ const puzzle = Array.from( {length: count + 1}, (_,i) => i === count ? [] : Array(cap).fill(i) ); let lastFrom = -1; let lastTo = -1; for(let step = 0; step < steps; step++){ const options = []; for(let from = 0; from < puzzle.length; from++){ if(puzzle[from].length === 0) continue; const colour = puzzle[from][puzzle[from].length - 1]; // Only move from single-colour stacks. if(!puzzle[from].every(c => c === colour)) continue; for(let to = 0; to < puzzle.length; to++){ if(to === from || puzzle[to].length >= cap) continue; // Avoid immediately reversing the last move. if(from === lastTo && to === lastFrom) continue; options.push([from,to]); } } if(options.length === 0) break; const [from,to] = options[ Math.floor(Math.random() * options.length) ]; puzzle[to].push(puzzle[from].pop()); lastFrom = from; lastTo = to; } const score = puzzleScore(puzzle); if(score > bestScore){ bestScore = score; bestPuzzle = puzzle.map(stack => stack.slice()); } } return bestPuzzle; }
